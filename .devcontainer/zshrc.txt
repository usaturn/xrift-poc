# 使い方: xrift-link [link|unlink] [dir]（既定は link と現在のディレクトリ）
function xrift-link(){
    local root="${XRIFT_POC_ROOT:-}"
    if [ -z "$root" ]; then
        print -u2 "xrift-link: XRIFT_POC_ROOT is not set"
        return 1
    fi
    local command="link"
    case "${1:-}" in
        link|unlink) command="$1"; shift ;;
    esac
    uv run --no-project python "${root}/.agents/scripts/project_link.py" "$command" "${1:-$PWD}"
}

# 起動時に使うワークスペースディレクトリを選択して標準出力へ出す。
# 1 件ならそのまま、0 件はルートを返す。複数件は fzf で選択する。
function herdr_select_workspace(){
    local root="${HERDR_WORKSPACE_ROOT:-/workspaces}"
    local -a dirs
    dirs=("$root"/*(N-/))
    case ${#dirs} in
        0)
            print -r -- "$root"
            ;;
        1)
            print -r -- "${dirs[1]}"
            ;;
        *)
            if ! command -v fzf >/dev/null 2>&1; then
                print -u2 "warning: fzf not found; using $root"
                print -r -- "$root"
                return 0
            fi
            local selected
            selected="$(printf '%s\n' "${dirs[@]}" \
                | fzf --height=40% --reverse --prompt='workspace> ')"
            if [ -n "$selected" ]; then
                print -r -- "$selected"
            else
                print -r -- "$root"
            fi
            ;;
    esac
}

# herdr はセッション復元時、session.json に保存済みの cwd を最優先し、クライアントが
# 渡す startup cwd を無視する（サーバログの "ignoring startup cwd"）。そのため一度
# ワークスペース外の cwd が保存されると、以後の復元でそこに固定される。
# $HERDR_WORKSPACE_ROOT 配下に実在しない cwd を $1 へ是正する。
# worktree 配下は AI エージェントのセッションが cwd に紐づくため保持する。
# サーバ稼働中は復元が起きず書き換えは競合するだけなので何もしない。
function herdr_repair_session_cwd(){
    local workspace_dir="$1"
    local session_file="${HOME}/.config/herdr/session.json"
    local root="${HERDR_WORKSPACE_ROOT:-/workspaces}"
    [ -n "$workspace_dir" ] || return 0
    # 修復先が実在し進入可能でなければ、壊れた cwd を別の壊れた cwd で置換することになるため何もしない
    [ -d "$workspace_dir" ] && [ -x "$workspace_dir" ] || return 0
    # fzf キャンセル等でルート自身が渡ると、焼き付いたあと inside(root,root) で自己修復不能になる
    [ "${workspace_dir%/}" != "${root%/}" ] || return 0
    [ -f "$session_file" ] || return 0
    command -v python3 >/dev/null 2>&1 || return 0
    local server_status
    server_status="$(herdr status server --json 2>/dev/null)" || return 0
    python3 - "$session_file" "$root" "$workspace_dir" "$server_status" <<'PY' || true
import json
import os
import sys

session_path, root, workspace, server_status = sys.argv[1:]

try:
    status = json.loads(server_status)
except (TypeError, ValueError):
    sys.exit(0)
if not isinstance(status, dict) or status.get("running") is not False:
    sys.exit(0)


def inside(candidate, base):
    """candidate が base 配下に実在するディレクトリなら True。"""
    if not isinstance(candidate, str) or not candidate:
        return False
    if not os.path.isdir(candidate):
        return False
    try:
        real_candidate = os.path.realpath(candidate)
        real_base = os.path.realpath(base)
        return os.path.commonpath([real_candidate, real_base]) == real_base
    except (ValueError, OSError):
        return False


try:
    with open(session_path, encoding="utf-8") as handle:
        data = json.load(handle)
except (OSError, ValueError):
    sys.exit(0)

# スキーマは herdr の内部形式。未知の version は構造が同じでも意味が
# 変わり得るため、無変更で何もしない
if not isinstance(data, dict) or data.get("version") != 3:
    sys.exit(0)

workspaces = data.get("workspaces")
if not isinstance(workspaces, list):
    sys.exit(0)

changed = False
# 修復先を既に identity_cwd に持つ workspace がある場合、それ以外のエントリは
# 書き換えない（pane を含むエントリ全体をスキップ）。複数エントリを同一
# identity_cwd へ収束させ、herdr が同一ワークスペースとして扱う事態を防ぐ。
target_taken = any(
    isinstance(entry, dict) and entry.get("identity_cwd") == workspace
    for entry in workspaces
)
for workspace_entry in workspaces:
    if not isinstance(workspace_entry, dict):
        continue
    identity = workspace_entry.get("identity_cwd")
    repair_identity = not inside(identity, root)
    if repair_identity:
        if target_taken:
            # 修復先占有時は identity も pane も触らない
            continue
        workspace_entry["identity_cwd"] = workspace
        identity = workspace
        target_taken = True
        changed = True
    # pane 是正先: 健全 identity ならその値、いま identity 修復したなら workspace
    pane_target = identity if inside(identity, root) else workspace
    tabs = workspace_entry.get("tabs")
    if not isinstance(tabs, list):
        continue
    for tab in tabs:
        if not isinstance(tab, dict):
            continue
        panes = tab.get("panes")
        if not isinstance(panes, dict):
            continue
        for pane in panes.values():
            if not isinstance(pane, dict):
                continue
            if not inside(pane.get("cwd"), root):
                pane["cwd"] = pane_target
                changed = True

if not changed:
    sys.exit(0)

tmp_path = session_path + ".tmp"
try:
    with open(tmp_path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2)
    os.replace(tmp_path, session_path)
except OSError:
    try:
        os.unlink(tmp_path)
    except OSError:
        pass
PY
}

function herdrstart(){
    if [ -n "${HERDR_SOCKET_PATH:-}" ] || [ "${HERDR_ENV:-}" = "1" ] \
        || [ ! -t 0 ] || [ ! -t 1 ]; then
        return 0
    fi
    local workspace_dir
    workspace_dir="$(herdr_select_workspace)"
    cd -- "$workspace_dir" 2>/dev/null \
        || print -u2 "warning: cannot cd $workspace_dir; staying in $PWD"
    export TZ="${TZ:-Asia/Tokyo}"
    if ! command -v herdr >/dev/null 2>&1; then
        print -u2 "warning: herdr not found; continuing without a multiplexer"
        return 0
    fi
    if command -v herdr-status-updater >/dev/null 2>&1; then
        herdr-status-updater --daemon || true
    fi
    herdr_repair_session_cwd "$workspace_dir"
    herdr
}

# ディスクを圧迫する使い捨て debris を間引いて掃除する（#146。詳細は
# cleanup-workspace.bash を参照）。
# 新しいシェルを開くたびに毎回実行すると重複起動になるため、
# 24 時間 (86400 秒) に 1 回へ間引く。手動での定期実行を要求しない。
function xrift_poc_periodic_cleanup(){
    command -v cleanup-workspace.bash >/dev/null 2>&1 || return 0
    local state_dir="${XRIFT_POC_CLEANUP_STATE_DIR:-$HOME/.cache/xrift-poc}"
    local stamp="${state_dir}/cleanup-workspace.stamp"
    mkdir -p "$state_dir" 2>/dev/null || return 0
    if [ -f "$stamp" ]; then
        local now epoch
        now=$(date +%s)
        epoch=$(date -r "$stamp" +%s 2>/dev/null) || return 0
        (( now - epoch >= 86400 )) || return 0
    fi
    # 複数シェルの同時起動でも二重実行しないよう、flock を使う。
    # mkdir ロックは overlayfs 上で排他にならず、空ディレクトリの rmdir 回収は
    # 実行中ロックも奪う。ロックはバックグラウンド子が保持し、親シェルは持たない。
    command -v flock >/dev/null 2>&1 || return 0
    # 旧 mkdir ロックが残した空ディレクトリがあると 9> が失敗する。
    if [ -d "${stamp}.lock" ]; then
        rmdir "${stamp}.lock" 2>/dev/null || return 0
    fi
    (
        flock -n 9 || exit 0
        touch "$stamp"
        cleanup-workspace.bash >"${state_dir}/cleanup-workspace.log" 2>&1
    ) 9>"${stamp}.lock" &!
}
xrift_poc_periodic_cleanup
