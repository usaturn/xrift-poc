"""指摘を JSON ファイルへ保存する。"""

import json
import re
from pathlib import Path

_FINDING_ID = re.compile(r"[A-Z]-[0-9]+")
_CONTROL = re.compile(r"[\x00-\x1f\x7f]")


def validate_finding(finding_id: str, content: str) -> None:
    """指摘の ID と本文を検査する。

    不正な入力は ValueError で拒否し、何も保存せずに止まる（安全側に倒れる）。
    本文が空のとき、または制御文字を含むときは拒否する。
    """
    if not _FINDING_ID.fullmatch(finding_id):
        raise ValueError(f"invalid finding id: {finding_id!r}")
    if not content.strip():
        raise ValueError("empty content")
    if _CONTROL.search(content):
        raise ValueError("content contains a control character")


def save_finding(store_dir: Path, finding_id: str, content: str) -> Path:
    """指摘を検査してから <store_dir>/<finding_id>.json に保存し、そのパスを返す。"""
    validate_finding(finding_id, content)
    path = store_dir / f"{finding_id}.json"
    path.write_text(json.dumps({"id": finding_id, "content": content}, ensure_ascii=False), encoding="utf-8")
    return path
